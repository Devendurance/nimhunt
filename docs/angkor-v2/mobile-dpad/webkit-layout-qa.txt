async (page) => {
  const results = [];
  for (const [width,height] of [[320,568],[360,640],[390,844],[430,932]]) {
    await page.setViewportSize({width,height});
    for (const scenario of ['stage1','stage3','transition']) {
      await page.goto(`http://127.0.0.1:5174/dev/angkor-v2-mobile-layout.html?scenario=${scenario}&reward=1&safe=34`);
      await page.waitForSelector('canvas');
      await page.waitForSelector('.v2-controls [role=group]');
      await page.getByText(/^Entering .*…$/).waitFor({state:'hidden',timeout:60000});
      await page.waitForTimeout(500);
      await page.waitForSelector('canvas');
      const metrics = await page.evaluate(() => {
        const shell = document.querySelector('.angkor-v2-play');
        const pad = document.querySelector('.v2-controls [role=group]');
        const rect = pad.getBoundingClientRect();
        const buttons = [...pad.querySelectorAll('button')].map(b=>({name:b.ariaLabel,...b.getBoundingClientRect().toJSON()}));
        const canvas = document.querySelector('canvas');
        return {width:innerWidth,height:visualViewport.height,center:rect.x+rect.width/2,padWidth:rect.width,buttons,safe:parseFloat(getComputedStyle(shell).paddingBottom),canvasHeight:canvas.height,displayHeight:canvas.getBoundingClientRect().height,overflow:document.documentElement.scrollWidth>innerWidth,selection:getComputedStyle(shell).userSelect,callout:getComputedStyle(shell).getPropertyValue('-webkit-touch-callout')};
      });
      if (Math.abs(metrics.center-width/2)>0.5 || metrics.overflow || metrics.displayHeight<80 || metrics.safe<34 || metrics.buttons.some(b=>b.y<0||b.bottom>height-34||b.width!==(width<=350?56:60)||b.height!==(width<=350?56:60))) throw Error(JSON.stringify({scenario,...metrics}));
      await page.screenshot({path:`output/playwright/v2-dpad-${width}-${scenario}-webkit.png`,scale:'css'});
      results.push({scenario,...metrics});
    }
  }
  return results;
}
